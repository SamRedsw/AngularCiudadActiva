import { HttpInterceptorFn } from '@angular/common/http';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = localStorage.getItem('jwt_token');
  if (token) {
    const cloned = req.clone({
      setHeaders: { Authorization: `Bearer ${token}` }
    });
    return next(cloned);
  }
  return next(req);
};

import { Component } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { Router } from '@angular/router';
import { ApiService } from '../../core/services/api.service';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './login.component.html',
  styleUrls: ['./login.component.scss']
})
export class LoginComponent {
  loginForm: FormGroup;
  isLoading = false;
  errorMessage: string | null = null;
  passwordVisible = false;

  demoAccounts = [
    { label: 'Ciudadano', email: 'ciudadano@ciudadactiva.gov.co', pass: 'Ciudadano123!' },
    { label: 'Gestor', email: 'gestor@ciudadactiva.gov.co', pass: 'Gestor123!' },
    { label: 'Operador', email: 'operador@ciudadactiva.gov.co', pass: 'Operador123!' },
    { label: 'Admin', email: 'admin@ciudadactiva.gov.co', pass: 'Admin123!' }
  ];

  constructor(
    private fb: FormBuilder,
    private apiService: ApiService,
    private router: Router
  ) {
    this.loginForm = this.fb.group({
      email: ['', [Validators.required, Validators.email]],
      password: ['', [Validators.required, Validators.minLength(6)]]
    });
  }

  selectDemoAccount(acc: { email: string; pass: string }) {
    this.loginForm.patchValue({
      email: acc.email,
      password: acc.pass
    });
  }

  togglePasswordVisibility() {
    this.passwordVisible = !this.passwordVisible;
  }

  onSubmit() {
    if (this.loginForm.invalid) return;

    this.isLoading = true;
    this.errorMessage = null;

    this.apiService.login(this.loginForm.value).subscribe({
      next: (res) => {
        this.isLoading = false;
        if (res.success && res.data) {
          const auth = res.data;
          localStorage.setItem('jwt_token', auth.token);
          localStorage.setItem('user_role', auth.rol || auth.usuario?.rol || 'ROLE_CIUDADANO');
          localStorage.setItem('user_name', auth.nombre || auth.usuario?.nombre || '');
          
          // RedirecciÃ³n por Rol
          const role = auth.rol || auth.usuario?.rol;
          if (role === 'ROLE_GESTOR') this.router.navigate(['/gestor/dashboard']);
          else if (role === 'ROLE_OPERADOR') this.router.navigate(['/operador/tareas']);
          else if (role === 'ROLE_ADMIN') this.router.navigate(['/admin/panel']);
          else this.router.navigate(['/ciudadano/home']);
        }
      },
      error: (err) => {
        this.isLoading = false;
        this.errorMessage = 'Credenciales invÃ¡lidas o problema de conexiÃ³n con el servidor.';
      }
    });
  }
}

import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { Router } from '@angular/router';
import { ApiService } from '../../core/services/api.service';
import { ReporteResponse, AsignacionModel, EstadisticasModel } from '../../core/models/models';

@Component({
  selector: 'app-gestor-dashboard',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './gestor-dashboard.component.html',
  styleUrls: ['./gestor-dashboard.component.scss']
})
export class GestorDashboardComponent implements OnInit {
  pendientes: ReporteResponse[] = [];
  asignaciones: AsignacionModel[] = [];
  stats: EstadisticasModel | null = null;
  reporteCritico: ReporteResponse | null = null;
  isLoading = true;

  constructor(private apiService: ApiService, private router: Router) {}

  ngOnInit(): void {
    this.cargarDatos();
  }

  cargarDatos(): void {
    this.isLoading = true;

    this.apiService.getReportes(undefined, 'PENDIENTE').subscribe(res => {
      if (res.success && res.data) {
        this.pendientes = res.data;
        this.reporteCritico = this.pendientes.find(r => r.prioridad === 'CRITICA') || this.pendientes[0] || null;
      }
    });

    this.apiService.getAsignacionesActivas().subscribe(res => {
      if (res.success && res.data) {
        this.asignaciones = res.data;
      }
    });

    this.apiService.getEstadisticas().subscribe(res => {
      if (res.success && res.data) {
        this.stats = res.data;
      }
      this.isLoading = false;
    });
  }

  inspeccionar(id: number): void {
    this.router.navigate(['/gestor/inspeccion', id]);
  }
}

import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ActivatedRoute, Router } from '@angular/router';
import { FormsModule } from '@angular/forms';
import { ApiService } from '../../core/services/api.service';
import { ReporteResponse, UsuarioModel } from '../../core/models/models';

@Component({
  selector: 'app-gestor-inspeccion',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './gestor-inspeccion.component.html',
  styleUrls: ['./gestor-inspeccion.component.scss']
})
export class GestorInspeccionComponent implements OnInit {
  reporteId!: number;
  reporte: ReporteResponse | null = null;
  operadores: UsuarioModel[] = [];
  prioridadSeleccionada = 'ALTA';
  
  // Modales y Reglas
  mostrarModalRechazo = false;
  motivoRechazo = '';
  mostrarModalAsignar = false;
  operadorSeleccionadoId: number | null = null;
  notasAsignacion = '';
  errorMessage: string | null = null;

  constructor(
    private route: ActivatedRoute,
    private router: Router,
    private apiService: ApiService
  ) {}

  ngOnInit(): void {
    this.reporteId = Number(this.route.snapshot.paramMap.get('id'));
    this.cargarDetalle();
  }

  cargarDetalle(): void {
    this.apiService.getReporteDetalle(this.reporteId).subscribe(res => {
      if (res.success && res.data) {
        this.reporte = res.data;
      }
    });

    this.apiService.getOperadores().subscribe(res => {
      if (res.success && res.data) {
        this.operadores = res.data;
      }
    });
  }

  // RN01: Rechazo con JustificaciÃ³n Obligatoria
  confirmarRechazo(): void {
    if (!this.motivoRechazo.trim()) {
      this.errorMessage = 'RN01: La justificaciÃ³n tÃ©cnica de rechazo es obligatoria.';
      return;
    }

    this.apiService.validarReporte(this.reporteId, {
      admitido: false,
      justificacion: this.motivoRechazo
    }).subscribe(res => {
      if (res.success) {
        this.router.navigate(['/gestor/dashboard']);
      }
    });
  }

  // RN02: PriorizaciÃ³n por Gestor
  admitirYPriorizar(): void {
    this.apiService.validarReporte(this.reporteId, {
      admitido: true,
      prioridad: this.prioridadSeleccionada,
      justificacion: 'Admitido oficialmente por Gestor'
    }).subscribe(res => {
      if (res.success) {
        this.mostrarModalAsignar = true;
      }
    });
  }

  confirmarAsignacion(): void {
    const opId = this.operadorSeleccionadoId || this.operadores[0]?.id;
    if (!opId) return;

    this.apiService.asignarOperador(this.reporteId, {
      operadorId: opId,
      notas: this.notasAsignacion
    }).subscribe(res => {
      if (res.success) {
        this.router.navigate(['/gestor/dashboard']);
      }
    });
  }
}
